static bool MoveJ_ReadFeedbackFromTopic(float q_rad[JOINT_NUM], float v_rad_s[JOINT_NUM])
{
    int i;
    uint32_t mask_got;

    /* 只要求 1/2/3 轴有反馈 */
    mask_got = algorithm_subscribe_arm_feedback_data.update_mask & ACTIVE_MASK;
    if (mask_got != ACTIVE_MASK)
    {
        return false;
    }

    /* 读取 1/2/3 轴真实反馈 */
    for (i = 0; i < ACTIVE_JOINT_NUM; i++)
    {
        q_rad[i]   = algorithm_subscribe_arm_feedback_data.joint[i].pos_rad;
        v_rad_s[i] = algorithm_subscribe_arm_feedback_data.joint[i].vel_rad_s;
    }

    /* 4/5/6 轴当前没有电机，固定为 0 */
    for (i = ACTIVE_JOINT_NUM; i < JOINT_NUM; i++)
    {
        q_rad[i]   = 0.0f;
        v_rad_s[i] = 0.0f;
    }

    return true;
}

static bool MoveJ_StartFromCurrentFeedback(void)
{
    float q_target_local[JOINT_NUM];

    if (!g_feedback_valid)
    {
        USART7_DebugPrintf("[MoveJ] feedback invalid\r\n");
        return false;
    }

    /* 先复制用户目标 */
    memcpy(q_target_local, g_q_target, sizeof(q_target_local));

    /* 4/5/6 轴强制保持当前位置 -> dq = 0，不参与规划 */
    q_target_local[3] = g_q_fb[3];
    q_target_local[4] = g_q_fb[4];
    q_target_local[5] = g_q_fb[5];

    if (!JointMoveJ_Start(&g_movej,
                          g_q_fb,
                          q_target_local,
                          g_vmax,
                          g_amax,
                          &g_joint_limit))
    {
        USART7_DebugPrintf("[MoveJ] start failed\r\n");
        return false;
    }

    USART7_DebugPrintf("[MoveJ] start ok, total_time=%.3f\r\n", g_movej.prof.total_time);
    return true;
}

bool AlgorithmTask_PostMoveJTargetRad(const float q_target_rad[JOINT_NUM])
{
    if (q_target_rad == 0)
    {
        return false;
    }

    taskENTER_CRITICAL();
    memcpy(g_q_target, q_target_rad, sizeof(g_q_target));
    g_movej_cmd_pending = 1;
    taskEXIT_CRITICAL();

    return true;
}

static float update_time_scale(const JointMoveJPlanner_t *jp,
                               const float q_fb[3],
                               const float v_fb[3],
                               float scale_prev)
{
    const float q_tol[3] = {0.03f, 0.03f, 0.03f};   // rad
    const float v_tol[3] = {0.20f, 0.20f, 0.20f};   // rad/s

    const float lambda = 0.3f;      // 速度误差权重
    const float k = 1.5f;           // 收缩强度
    const float scale_min = 0.15f;  // 最小时间缩放
    const float alpha = 0.10f;      // 一阶滤波系数

    float eq_n = 0.0f;
    float ev_n = 0.0f;
    float z, scale_raw, scale;

    for (int i = 0; i < 3; i++)
    {
        float eq = fabsf(jp->q_ref[i] - q_fb[i]) / q_tol[i];
        float ev = fabsf(jp->v_ref[i] - v_fb[i]) / v_tol[i];

        if (eq > eq_n) eq_n = eq;
        if (ev > ev_n) ev_n = ev;
    }

    z = eq_n + lambda * ev_n;

    scale_raw = scale_min + (1.0f - scale_min) / (1.0f + k * z);

    scale = scale_prev + alpha * (scale_raw - scale_prev);

    if (scale < scale_min) scale = scale_min;
    if (scale > 1.0f)      scale = 1.0f;

    return scale;
}
static float g_scale = 1.0f;

/* -------------------------------- 线程入口 ------------------------------- */
void AlgorithmTask_Entry(void const * argument)
{
/* -------------------------------- 外设初始化段落 ------------------------------- */
    Init_KalmanFiltersOne(KALMAN_F, KALMAN_H, KALMAN_Q, KALMAN_R);
    /* MoveJ 初始化：只做一次 */
    JointMoveJ_Init(&g_movej);

    for (uint16_t i = 0; i < JOINT_NUM; i++)
    {
        g_joint_limit.min[i] = -3.1415926f;
        g_joint_limit.max[i] =  3.1415926f;
    }

/* -------------------------------- 外设初始化段落 ------------------------------- */

/* -------------------------------- 线程间Topics初始化 ------------------------------- */
    algorithm_topic_sub_init();
    algorithm_topic_pub_init();
/* -------------------------------- 线程间Topics初始化 ------------------------------- */
/* -------------------------------- 调试监测线程调度 --------------------------------- */
    algorithm_task_dt = dwt_get_delta(&algorithm_task_dwt);
    algorithm_task_start_dt = dwt_get_time_ms();
/* -------------------------------- 调试监测线程调度 --------------------------------- */
    for(;;)
    {
/* -------------------------------- 调试监测线程调度 --------------------------------- */
        algorithm_task_delta = dwt_get_time_ms() - algorithm_task_start_dt;
        algorithm_task_start_dt = dwt_get_time_ms();
        algorithm_task_dt = dwt_get_delta(&algorithm_task_dwt);
/* -------------------------------- 调试监测线程调度 --------------------------------- */
/* -------------------------------- 线程订阅Topics信息 ------------------------------- */
        algorithm_topic_sub_pull();
/* -------------------------------- 线程订阅Topics信息 ------------------------------- */

/* -------------------------------- 线程代码编写段落 ------------------------------- */

        if (xQueueReceive(xKalmanOneQueue, angles, 0) == pdTRUE) {
            // 对接收的数据进行滤波处理
            KalmanFilterOne_Data(angles, filtered_data);
            xQueueSend(xControlQueue, filtered_data, 0);
        }

        if (MoveJ_ReadFeedbackFromTopic(g_q_fb, g_v_fb))
        {
            g_feedback_valid = 1;
        }

        /* -------------------------------- 收到新目标时，只启动一次 --------------------------------- */
        if (g_movej_cmd_pending)
        {
            if (MoveJ_StartFromCurrentFeedback())
            {
                g_movej_cmd_pending = 0;
            }
        }

        /* 真正的周期更新：每次只走一步 */
        if (g_movej.state == JP_RUNNING)
        {
            float dt_real = algorithm_task_dt;   // 如果 dwt_get_delta 返回 ms
            float dt_eff;
            g_scale = update_time_scale(&g_movej, g_q_fb, g_v_fb, g_scale);
            dt_eff = dt_real * g_scale;

            if (!JointMoveJ_Update(&g_movej, dt_eff))
            {
                USART7_DebugPrintf("[MoveJ] Update failed\r\n");
                JointMoveJ_Stop(&g_movej);
            }
            else
            {
                /* 这里把 q_ref / v_ref 下发给控制器 */
                /* 例如：Control_SetJointRef(g_movej.q_ref, g_movej.v_ref); */
            }
        }
        else if (g_movej.state == JP_DONE)
        {
            /* 到位后保持终点参考，或者置空闲 */
            /* g_movej.state = JP_IDLE;  // 若你希望任务完成后恢复空闲 */
        }
        else if (g_movej.state == JP_FAULT)
        {
            USART7_DebugPrintf("[MoveJ] planner fault\r\n");
        }

/* -------------------------------- 线程代码编写段落 ------------------------------- */

/* -------------------------------- 线程发布Topics信息 ------------------------------- */
        algorithm_topic_pub_push();
/* -------------------------------- 线程发布Topics信息 ------------------------------- */
        vTaskDelay(1);
    }
}
/* -------------------------------- 线程结束 ------------------------------- */

/* -------------------------------- 线程间通讯Topics相关 ------------------------------- */
static void algorithm_topic_pub_init(void)
{
    publish_movej_ref_topic = pub_register("movej_ref_pub", sizeof(movej_ref_msg_t));
}

static void algorithm_topic_sub_init(void)
{
    // 订阅关节反馈数据
    subscribe_arm_feedback_topic = sub_register("dm_arm_feedback_pub", sizeof(dm_arm_feedback_msg_t));
}

static void algorithm_topic_pub_push(void)
{
    int i;

    memset(&algorithm_publish_movej_ref_data, 0, sizeof(algorithm_publish_movej_ref_data));

    algorithm_publish_movej_ref_data.seq = ++g_movej_pub_seq;
    algorithm_publish_movej_ref_data.tick_ms = algorithm_subscribe_arm_feedback_data.tick_ms;
    algorithm_publish_movej_ref_data.planner_state = (uint8_t)g_movej.state;

    if ((g_movej.state == JP_RUNNING) || (g_movej.state == JP_DONE))
    {
        algorithm_publish_movej_ref_data.valid = 1;

        for (i = 0; i < JOINT_NUM; i++)
        {
            algorithm_publish_movej_ref_data.q_ref_rad[i]   = g_movej.q_ref[i];
            algorithm_publish_movej_ref_data.v_ref_rad_s[i] = g_movej.v_ref[i];

            algorithm_publish_movej_ref_data.q_fb_rad[i]    = g_q_fb[i];
            algorithm_publish_movej_ref_data.v_fb_rad_s[i]  = g_v_fb[i];
        }
    }
    else
    {
        algorithm_publish_movej_ref_data.valid = 0;
    }

    if (!pub_push_msg(publish_movej_ref_topic, &algorithm_publish_movej_ref_data))
    {
        USART7_DebugPrintf("[MoveJ] topic push failed\r\n");
    }
}

static void algorithm_topic_sub_pull(void)
{
    sub_get_msg(subscribe_arm_feedback_topic, &algorithm_subscribe_arm_feedback_data);
}
